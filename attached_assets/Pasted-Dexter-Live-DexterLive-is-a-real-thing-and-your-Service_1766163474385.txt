Query by brand/code/symptom → output repair plan, parts, links. Run on Replit (add XAI_API_KEY in .env). Demo to Ryan: "See how it pulls Dexter codes + AAdvantage parts links in 10sec."

```python
from flask import Flask, request, jsonify
import os
from openai import OpenAI  # Grok-compatible

app = Flask(__name__)
client = OpenAI(api_key=os.getenv("XAI_API_KEY"), base_url="https://api.x.ai/v1")

# Database from above JSON (paste the brands dict here)
database = { "brands": { ... } }  # Paste the full JSON above

@app.route('/diagnose', methods=['POST'])
def diagnose():
    data = request.json
    brand = data.get('brand')
    code = data.get('code')
    symptom = data.get('symptom')

    if brand in database['brands']:
        brand_data = database['brands'][brand]
        for err in brand_data['error_codes']:
            if err['code'] == code:
                return jsonify(err)
    
    # Grok fallback for unknown
    prompt = f"Diagnose {brand} commercial laundry error {code} or symptom '{symptom}': Provide troubleshooting, common parts with numbers/links, repair steps."
    response = client.chat.completions.create(model="grok-4", messages=[{"role": "user", "content": prompt}])
    return jsonify({"ai_response": response.choices[0].message.content})

if __name__ == '__main__':
    app.run(debug=True)
