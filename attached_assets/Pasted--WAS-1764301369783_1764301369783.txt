const getOptimalPrice = (machine: Machine, storeData: StoreData) => {
  const basePrice = machine.capacity === 20 ? 3.5 : machine.capacity === 60 ? 6.5 : 8.0;
  const competitionFactor = storeData.competitorsWithin2Miles > 3 ? 0.9 : 1.15;
  const peakHourBoost = isPeakHour() ? 1.25 : 1.0;
  const demandMultiplier = storeData.currentOccupancy > 0.8 ? 1.3 : 1.0;
  
  return roundToQuarter(basePrice * competitionFactor * peakHourBoost * demandMultiplier);
};
