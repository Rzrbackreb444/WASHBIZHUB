tsx// src/pages/CleanbiCalc.tsx
import React, { useState } from 'react';
import { Link } from 'react-router-dom';

const calculateCLEANBI = (inputs) => {
  const { rentersPerMile = 0, tpd = 0, noi = 0, hasWDF = false } = inputs;
  let score = 0;
  if (rentersPerMile > 1500) score += 20;
  else if (rentersPerMile > 1000) score += 15;
  if (tpd >= 5.5) score += 20;
  if (hasWDF) score += 15;
  if (noi > 100000) score += 20;
  const grade = score >= 90 ? 'A' : score >= 75 ? 'B' : score >= 60 ? 'C' : 'Needs Work';
  return { total: score, grade };
};

export default function CleanbiCalc() {
  const [inputs, setInputs] = useState({ rentersPerMile: 1200, tpd: 4.5, noi: 100000, hasWDF: true });
  const [result, setResult] = useState(null);

  const run = () => setResult(calculateCLEANBI(inputs));

  return (
    <div className="min-h-screen bg-gradient-to-br from-gray-900 to-black text-white py-32 text-center">
      <h1 className="text-6xl font-black mb-8">CLEANBI™ Score</h1>
      <div className="max-w-md mx-auto bg-white/10 p-8 rounded-3xl">
        <input type="number" placeholder="Renters/Mile" value={inputs.rentersPerMile} onChange={e => setInputs({...inputs, rentersPerMile: +e.target.value})} className="w-full p-4 mb-4 bg-white/20 rounded-lg" />
        <input type="number" placeholder="TPD" value={inputs.tpd} onChange={e => setInputs({...inputs, tpd: +e.target.value})} className="w-full p-4 mb-4 bg-white/20 rounded-lg" />
        <input type="number" placeholder="NOI ($)" value={inputs.noi} onChange={e => setInputs({...inputs, noi: +e.target.value})} className="w-full p-4 mb-4 bg-white/20 rounded-lg" />
        <select value={inputs.hasWDF} onChange={e => setInputs({...inputs, hasWDF: e.target.value === 'true'})} className="w-full p-4 mb-8 bg-white/20 rounded-lg">
          <option value={true}>Has WDF</option>
          <option value={false}>No WDF</option>
        </select>
        <button onClick={run} className="w-full py-4 bg-accent text-primary rounded-full font-bold">
          Calculate
        </button>
        {result && (
          <div className="mt-8">
            <p className="text-4xl font-black">{result.total} ({result.grade})</p>
            <Link to="/report" className="mt-4 inline-block px-8 py-4 border-2 border-accent text-accent rounded-full font-bold">
              Full Report $297
            </Link>
          </div>
        )}

}
