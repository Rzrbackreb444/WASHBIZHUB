const calculateCLEANBI = (inputs) => {
  const {
    rentersPerMile = 0, medianIncome = 0, parkingRatio = 0, visibilityScore = 0,
    leaseYears = 0, equipmentAge = 0, equipmentBrand = '', uptimePercent = 0,
    hasWDF = false, hasPUD = false, hasATM = false, cashlessPercent = 0,
    tpd = 0, ebitdaMargin = 0, dscr = 0, roi = 0,
    rivalsPerMile = 0, competitorPricing = 0, googleStars = 0,
    hasStaffing = false, hasMarketing = false
  } = inputs;

  let customerScore = 0;
  if (rentersPerMile > 1500) customerScore += 10;
  else if (rentersPerMile > 1000) customerScore += 7;
  else if (rentersPerMile > 500) customerScore += 4;
  if (medianIncome > 65000) customerScore += 5;
  else if (medianIncome > 45000) customerScore += 3;
  else if (medianIncome > 30000) customerScore += 1;

  let locationScore = 0;
  if (parkingRatio >= 1.5) locationScore += 8;
  else if (parkingRatio >= 1.0) locationScore += 5;
  locationScore += visibilityScore;
  if (leaseYears >= 10) locationScore += 2;

  let equipScore = 0;
  if (equipmentAge < 5) equipScore += 8;
  else if (equipmentAge < 10) equipScore += 5;
  else if (equipmentAge < 15) equipScore += 2;
  if (['Dexter', 'Girbau'].includes(equipmentBrand)) equipScore += 4;
  if (uptimePercent >= 99) equipScore += 3;

  let adaptScore = 0;
  if (hasWDF) adaptScore += 6;
  if (hasPUD) adaptScore += 3;
  if (hasATM) adaptScore += 3;
  if (cashlessPercent >= 70) adaptScore += 3;

  let numbersScore = 0;
  if (tpd >= 5.5) numbersScore += 6;
  else if (tpd >= 4.5) numbersScore += 4;
  else if (tpd >= 3.5) numbersScore += 2;
  if (ebitdaMargin >= 25) numbersScore += 4;
  else if (ebitdaMargin >= 20) numbersScore += 2;
  if (dscr >= 1.25) numbersScore += 3;
  if (roi >= 20) numbersScore += 2;

  let intelScore = 0;
  if (rivalsPerMile <= 2) intelScore += 5;
  else if (rivalsPerMile <= 3) intelScore += 3;
  if (competitorPricing < 3.50) intelScore += 3;
  if (googleStars >= 4.5) intelScore += 2;

  let brandScore = 0;
  if (hasStaffing) brandScore += 2;
  if (hasMarketing) brandScore += 3;

  const total = 
    (customerScore * 0.20) +
    (locationScore * 0.20) +
    (equipScore * 0.15) +
    (adaptScore * 0.15) +
    (numbersScore * 0.15) +
    (intelScore * 0.10) +
    (brandScore * 0.05);

  const grade = total >= 90 ? 'A' : total >= 75 ? 'B' : total >= 60 ? 'C' : 'Needs Work';

  return { total: total.toFixed(1), grade, breakdown: { customerScore, locationScore, equipScore, adaptScore, numbersScore, intelScore, brandScore } };
};

// Design Studio Component
function DesignStudio() {
  const [layout, setLayout] = useState([]);
  const [total, setTotal] = useState(0);

  const addItem = (item) => {
    const newItem = { ...item, id: Date.now(), x: 100, y: 100 };
    setLayout([...layout, newItem]);
    setTotal(total + item.price);
  };

  return (
    <div className="min-h-screen bg-gradient-to-br from-gray-900 to-black text-white p-8">
      <h1 className="text-5xl font-black mb-8">Design Studio</h1>
      <div className="grid md:grid-cols-4 gap-8">
        <div className="bg-white/10 p-6 rounded-xl">
          <h2 className="text-xl font-bold mb-4">Add Items</h2>
          {[
            { name: 'Dexter T-900', price: 8500 },
            { name: 'Vending', price: 1200 },
          ].map(item => (
            <button key={item.name} onClick={() => addItem(item)} className="w-full p-3 bg-accent text-primary rounded-lg mb-2">
              {item.name} — ${item.price}
            </button>
          ))}
        </div>
        <div className="md:col-span-3 bg-white/10 p-6 rounded-xl">
          <Stage width={900} height={600}>
            <Layer>
              {layout.map(item => (
                <Rect key={item.id} x={item.x} y={item.y} width={50} height={50} fill="#3b82f6" />
              ))}
            </Layer>
          </Stage>
          <p className="mt-4 text-xl">Total: ${total.toLocaleString()}</p>
        </div>
      </div>
    </div>
  );
}

// Calculator Component
function Calc() {
  const [inputs, setInputs] = useState({ machines: 20, tpd: 8, price: 5 });
  const monthly = inputs.machines * inputs.tpd * inputs.price * 30;
  const annual = monthly * 12;

  return (
    <div className="min-h-screen bg-gradient-to-br from-gray-900 to-black text-white py-32 text-center">
      <h1 className="text-6xl font-black mb-8">Laundromat Calculator</h1>
      <div className="max-w-md mx-auto bg-white/10 p-8 rounded-3xl">
        <input type="number" placeholder="Machines" value={inputs.machines} onChange={e => setInputs({...inputs, machines: +e.target.value})} className="w-full p-4 mb-4 bg-white/20 rounded-lg" />
        <input type="number" placeholder="TPD" value={inputs.tpd} onChange={e => setInputs({...inputs, tpd: +e.target.value})} className="w-full p-4 mb-4 bg-white/20 rounded-lg" />
        <input type="number" placeholder="Price/Wash" value={inputs.price} onChange={e => setInputs({...inputs, price: +e.target.value})} className="w-full p-4 mb-8 bg-white/20 rounded-lg" />
        <p className="text-4xl font-black text-accent">${annual.toLocaleString()}/yr</p>
      </div>
    </div>
  );
}
