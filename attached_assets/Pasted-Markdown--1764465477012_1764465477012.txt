import { Radar } from 'react-chartjs-2';

  LineElement,
  Filler,
  Tooltip,
  Legend,
} from 'chart.js';

ChartJS.register(RadialLinearScale, PointElement, LineElement, Filler, Tooltip, Legend);

function App() {
  const [address, setAddress] = useState('');
  const [showCleanbi, setShowCleanbi] = useState(false);

  const runDemo = () => address.trim() && setShowCleanbi(true);

  const radarData = {
    labels: ['Population','Income','Renters','Age','Competition','Traffic','Visibility','Sq Ft','Machines','Parking','Equip Age','Cleanliness','Pricing','Hours','Drop-Off','Card System','Reviews'],
    datasets: [{

      backgroundColor: 'rgba(57,204,204,0.2)',
      borderColor: '#39CCCC',
      borderWidth: 4,
      pointBackgroundColor: '#39CCCC',
    }],
  };

  const radarOptions = {
    scales: { r: { min: 0, max: 10, ticks: { stepSize: 2 }, grid: { color: 'rgba(255,255,255,0.1)' }, pointLabels: { color: '#fff', font: { size: 14 } } } },
    plugins: { legend: { display: false } },
  };

  return (
    <>
      {/* STICKY HEADER */}
      <header className="fixed top-0 left-0 right-0 z-50 bg-navy-900/95 backdrop-blur-lg border-b border-teal-400/20">
        <div className="max-w-7xl mx-auto px-6 py-4 flex justify-between items-center">
          <h1 className="text-4xl font-bold text-teal-400 font-bebas">WashBizHub</h1>
          <div className="flex gap-8 items-center">
            <a href="#cleanbi" className="text-white hover:text-teal-400 transition text-lg">CLEANBI</a>
            <a href="/pos" className="text-white hover:text-teal-400 transition text-lg">POS</a>
            <a href="/design-studio" className="text-white hover:text-teal-400 transition text-lg">Design Studio</a>
            <a href="/trial" className="bg-teal-400 text-navy-900 px-10 py-4 rounded-xl font-bold hover:scale-105 transition text-lg">
              14-Day Free Trial
            </a>
          </div>
        </div>
      </header>

      {/* HERO + CLEANBI DEMO */}
      <main className="pt-24 min-h-screen">
        <section className="max-w-7xl mx-auto px-6 text-center py-20">
          <h1 className="text-5xl md:text-7xl font-bold font-bebas leading-tight">
            The Only Laundromat Platform That <span className="text-teal-400">Pays You Back</span> Before You Pay Us
          </h1>
          <p className="text-2xl mt-6 text-gray-200">
            AI predicts failures • Dynamic pricing adds $24k/year • Marketplace gives you cash back
          </p>
          <p className="text-xl mt-4 text-gray-300">Join 72,000+ owners already winning →</p>

          <div id="cleanbi" className="mt-20 bg-white/10 backdrop-blur-xl rounded-3xl p-12 max-w-5xl mx-auto">
            {!showCleanbi ? (
              <>
                <h2 className="text-5xl font-bebas mb-10">Enter Any Address → See Revenue in 8 Seconds</h2>
                <input
                  type="text"
                  placeholder="e.g. 1234 Main St, Dallas, TX"
                  className="w-full max-w-3xl px-10 py-8 text-2xl text-navy-900 rounded-2xl mb-10 focus:outline-none focus:ring-4 focus:ring-teal-400"

                  onChange={(e) => setAddress(e.target.value)}

                    Download Full Report
                  </a>
                </div>
              </div>
            )}
          </div>
        </section>
      </main>
    </>
  );
}
