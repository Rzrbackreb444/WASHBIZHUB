function App() {
  const [items, setItems] = useState([]); // Machines + everything
  const [revenue, setRevenue] = useState(0);
  const [cleanbi, setCleanbi] = useState(0);
  const [price, setPrice] = useState(3.5);
  const [constructionCost, setConstructionCost] = useState(0);

  const addItem = (item) => {
    const newItem = { ...item, pos: [Math.random() * 10, 0, Math.random() * 10] };
    setItems([...items, newItem]);
    if (newItem.type === 'washer' || newItem.type === 'dryer') {
      const tpd = (18 * 60 / newItem.specs.cycle) * 0.45; // Turns Per Day = (hours * min/hour / cycle) * utilization
      const monthlyRev = tpd * price * 30 * (newItem.type === 'dryer' ? 0.7 : 1); // Dryers 70% of washer rev
      setRevenue(prev => prev + monthlyRev);
      setCleanbi(prev => Math.min(100, prev + (newItem.capacity / 10)));
    } else if (newItem.type === 'bulkhead') {
      setConstructionCost(prev => prev + (newItem.dimensions.w * newItem.dimensions.h / 144 * 20)); // $20/sq ft for bulkheads
    } else if (newItem.type === 'atm') {
      setRevenue(prev => prev + 1200); // $1,200/mo avg ATM fees
    } // Add more for POS (+$500/mo), changers (+$800/mo), dog wash (+$2,000/mo), video games (+$500/mo), carts/tables (+5% utilization = +revenue)
  };

  const optimizePricing = () => {
    const newPrice = price * (items.filter(i => i.type === 'washer').length > 0 ? 1.25 : 1.0); // 25% boost if washers present
    setPrice(newPrice);
    setRevenue(prev => prev * 1.25);
  };

  const exportPDF = () => {
    html2pdf().from(document.getElementById('export-area')).save('washbizhub-design.pdf');
  };

  const radarData = {
    labels: ['Capacity', 'Revenue', 'CLEANBI', 'Pricing', 'ROI'],
    datasets: [{
      label: 'Your Store',
      data: [items.length * 10, revenue / 10000, cleanbi, price * 10, revenue / 10000 * 1.2],
      backgroundColor: 'rgba(57,204,204,0.2)',
      borderColor: '#39CCCC',
      borderWidth: 4,
    }],
  };

  const radarOptions = { scales: { r: { min: 0, max: 100 } }, plugins: { legend: { display: false } } };

  return (
    <>
      <header className="fixed top-0 left-0 right-0 z-50 bg-navy-900/95 backdrop-blur border-b border-teal-400/30">
        <div className="max-w-7xl mx-auto px-6 py-4 flex justify-between items-center">
          <h1 className="text-4xl font-bold text-teal-400 font-bebas">WashBizHub Design Studio</h1>
          <div className="flex gap-6">
            <button onClick={optimizePricing} className="text-white hover:text-teal-400">Optimize Pricing</button>
            <button onClick={exportPDF} className="text-white hover:text-teal-400">Export PDF</button>
            <a href="/trial" className="bg-teal-400 text-navy-900 px-8 py-3 rounded-lg font-bold">Launch in POS</a>
          </div>
        </div>
      </header>

      <div className="flex h-screen pt-20">
        {/* 3D Canvas */}
        <div className="flex-1">
          <Canvas>
            <PerspectiveCamera makeDefault position={[20, 20, 20]} />
            <OrbitControls />
            <ambientLight intensity={0.8} />
            <directionalLight position={[10, 10, 5]} />
            <Grid infiniteGrid fadeDistance={50} />
            {items.map((item, i) => {
              const { scene } = useGLTF(item.glb); // Real GLB models from TurboSquid/Blender Market
              return <primitive key={i} object= {scene} position={item.pos} scale={[0.5, 0.5, 0.5]} />; // Scaled for realism
            })}
          </Canvas>
        </div>

        {/* Sidebar */}
        <div className="w-96 bg-navy-800 p-8 overflow-y-auto">
          <h2 className="text-3xl font-bebas text-teal-400 mb-8">Drag Items In</h2>
          <div className="grid grid-cols-1 gap-6">
            <button onClick={() => addItem(itemsLibrary[0])} className="bg-teal-400/20 p-6 rounded-xl hover:bg-teal-400/40 transition">
              {itemsLibrary[0].brand} {itemsLibrary[0].model} ({itemsLibrary[0].capacity} lb) · +${itemsLibrary[0].specs.energy * 10}/mo
            </button>
            <button onClick={() => addItem(itemsLibrary[1])} className="bg-teal-400/20 p-6 rounded-xl hover:bg-teal-400/40 transition">
              {itemsLibrary[1].brand} {itemsLibrary[1].model} ({itemsLibrary[1].capacity} lb) · +${itemsLibrary[1].specs.energy * 10}/mo
            </button>
            <button onClick={() => addItem(itemsLibrary[501])} className="bg-teal-400/20 p-6 rounded-xl hover:bg-teal-400/40 transition">
              Bulkhead (4x8 ft) · +$500 construct
            </button>
            <button onClick={() => addItem(itemsLibrary[502])} className="bg-teal-400/20 p-6 rounded-xl hover:bg-teal-400/40 transition">
              Hyosung ATM · +$1,200/mo fees
            </button>
            <button onClick={() => addItem(itemsLibrary[503])} className="bg-teal-400/20 p-6 rounded-xl hover:bg-teal-400/40 transition">
              Wash-Dry-Fold POS · +$500/mo
            </button>
            <button onClick={() => addItem(itemsLibrary[504])} className="bg-teal-400/20 p-6 rounded-xl hover:bg-teal-400/40 transition">
              American Changer · +$800/mo
            </button>

              R&B Wire Cart · +$249
            </button>
            <button onClick={() => addItem(itemsLibrary[508])} className="bg-teal-400/20 p-6 rounded-xl hover:bg-teal-400/40 transition">
              Hercules Bench · +$229
            </button>

              Stainless Table · +$209
            </button>

          </div>

          <div className="mt-12 bg-teal-400/10 rounded-2xl p-8 text-center" id="export-area">
            <h3 className="text-4xl font-bold">Revenue: ${revenue.toLocaleString()}/mo</h3>

}
