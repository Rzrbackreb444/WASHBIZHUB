// Room Dimensions (ft → scaled)
const SCALE = 0.1; // 1 unit = 1 ft
const ROOM = { width: 36, height: 12, depth: 24 };

// Machine/Furniture Models
const MODELS = {
  'Dexter T-900': { width: 3, depth: 4, height: 6, color: '#1e40af', price: 8500 },
  'Speed Queen Stack': { width: 2.5, depth: 3, height: 7, color: '#dc2626', price: 6500 },
  'Folding Table': { width: 6, depth: 2.5, height: 3, color: '#f59e0b', price: 300 },
  'Vending': { width: 3, depth: 2, height: 6, color: '#8b5cf6', price: 1200 },
  'ATM': { width: 2, depth: 2.5, height: 5, color: '#ef4444', price: 2500 },
};

// 3D Item with Drag
function DraggableItem({ item, onDrag, isSelected, onSelect }) {
  const mesh = useRef();
  const [hovered, setHovered] = useState(false);

  return (
    <group 
      position={[item.x * SCALE, item.height * SCALE / 2, item.z * SCALE]} 
      onClick={() => onSelect(item.id)}
      onPointerOver={() => setHovered(true)}
      onPointerOut={() => setHovered(false)}
    >
      <Box
        ref={mesh}
        args={[item.width * SCALE, item.height * SCALE, item.depth * SCALE]}
        onDrag={onDrag}
      >
        <meshStandardMaterial color={hovered || isSelected ? '#fbbf24' : item.color} />
      </Box>
      <Text
        position={[0, item.height * SCALE + 0.5, 0]}
        fontSize={0.6}
        color="#fff"
        anchorX="center"
        anchorY="middle"
      >
        {item.name}
      </Text>
    </group>
  );
}

// Main 3D Studio
export default function DesignStudio3D() {
  const [layout, setLayout] = useState([
    { id: 1, name: 'Dexter T-900', x: 5, z: 5, width: 3, depth: 4, height: 6, color: '#1e40af', price: 8500 },
    { id: 2, name: 'Vending', x: 15, z: 10, width: 3, depth: 2, height: 6, color: '#8b5cf6', price: 1200 },
  ]);
  const [selectedId, setSelectedId] = useState(null);
  const [total, setTotal] = useState(9700);

    const newX = data.point.x / SCALE;
    const newZ = data.point.z / SCALE;
    setLayout(layout.map(item => 
      item.id === id ? { ...item, x: newX, z: newZ } : item
    ));
  };

      ...model,
      id: Date.now(),
      x: 10 + Math.random() * 10,
      z: 10 + Math.random() * 10,

    setLayout([...layout, newItem]);
    setTotal(total + model.price);
  };

  return (
    <>
      <Helmet>
        <title>3D Design Studio Pro | WashBizHub</title>
      </Helmet>

      <div className="min-h-screen bg-gradient-to-br from-gray-900 via-gray-800 to-black text-white">

                2D Version

                Unlock Pro <Lock className="w-5 h-5" />
              </Link>
            </div>
          </div>
        </section>

        <section className="py-20">
          <div className="max-w-7xl mx-auto px-4">

                  <Canvas camera={{ position: [30, 30, 30], fov: 60 }}>
