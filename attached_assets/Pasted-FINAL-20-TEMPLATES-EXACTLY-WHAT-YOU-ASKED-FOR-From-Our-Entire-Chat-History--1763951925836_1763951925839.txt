tsx// app/vault/financials/page.tsx — FULLY INTERACTIVE 5-YEAR FINANCIAL MODEL
'use client';

import { useState } from 'react';
import { motion } from 'framer-motion';

export default function Financials() {
  const [inputs, setInputs] = useState({
    revenue: 240000,
    rent: 36000,
    utilities: 24000,
    labor: 48000,
    supplies: 18000,
    wdfRevenue: 120000,
    wdfLabor: 60000,
    capex: 20000,
    growthRate: 15,
  });

  const calculateYear = (year: number) => {
    const growth = Math.pow(1 + inputs.growthRate / 100, year - 1);
    const revenue = inputs.revenue * growth + inputs.wdfRevenue * growth;
    const expenses = inputs.rent * 1.03 ** (year - 1) +
      inputs.utilities * 1.05 ** (year - 1) +
      inputs.labor * 1.06 ** (year - 1) +
      inputs.supplies * 1.04 ** (year - 1) +
      inputs.wdfLabor * growth +
      inputs.capex;

    const profit = revenue - expenses;
    const margin = revenue > 0 ? (profit / revenue) * 100 : 0;

    return { revenue: revenue.toFixed(0), expenses: expenses.toFixed(0), profit: profit.toFixed(0), margin: margin.toFixed(1) };
  };

  const years = [2025, 2026, 2027, 2028, 2029].map(year => ({
    year,
    ...calculateYear(year - 2024)
  }));

  return (
    <div className="min-h-screen py-20 bg-gradient-to-br from-navy via-green-900 to-navy">
      <div className="max-w-7xl mx-auto px-6">
        <h1 className="text-7xl font-black text-center mb-16 text-yellow-400">5-Year Financial Pro Forma</h1>

        <div className="grid md:grid-cols-4 gap-8 mb-16">
          <input type="number" placeholder="Year 1 Revenue" value={inputs.revenue} onChange={e => setInputs({ ...inputs, revenue: +e.target.value })} className="p-8 text-3xl rounded-2xl text-black" />
          <input type="number" placeholder="Rent" value={inputs.rent} onChange={e => setInputs({ ...inputs, rent: +e.target.value })} className="p-8 text-3xl rounded-2xl text-black" />
          <input type="number" placeholder="WDF Revenue" value={inputs.wdfRevenue} onChange={e => setInputs({ ...inputs, wdfRevenue: +e.target.value })} className="p-8 text-3xl rounded-2xl text-black" />
          <input type="number" placeholder="Growth Rate %" value={inputs.growthRate} onChange={e => setInputs({ ...inputs, growthRate: +e.target.value })} className="p-8 text-3xl rounded-2xl text-black" />
        </div>

        <div className="bg-white/10 backdrop-blur-xl rounded-3xl p-16 overflow-x-auto">
          <table className="w-full text-2xl min-w-max">
            <thead>
              <tr className="border-b-4 border-yellow-400">
                <th className="text-left py-6">Year</th>
                <th className="text-right py-6">Revenue</th>
                <th className="text-right py-6">Expenses</th>
                <th className="text-right py-6">Profit</th>
                <th className="text-right py-6">Margin</th>
              </tr>
            </thead>
            <tbody>
              {years.map(y => (
                <tr key={y.year} className="border-b border-white/10">
                  <td className="py-8 font-bold">{y.year}</td>
                  <td className="text-right text-green-400">${parseInt(y.revenue).toLocaleString()}</td>
                  <td className="text-right text-red-400">${parseInt(y.expenses).toLocaleString()}</td>
                  <td className="text-right text-yellow-400">${parseInt(y.profit).toLocaleString()}</td>

    revenue: 240000,
    machines: 50,
    rent: 36000,
    utilities: 24000,
    labor: 48000,
    ownerSalary: 80000,
    wdfRevenue: 120000,
  });

  const [results, setResults] = useState<any>(null);

  const runMonteCarlo = () => {
    const simulations = 10000;
    const scores = [];

    for (let i = 0; i < simulations; i++) {
      const rev = inputs.revenue * (0.75 + Math.random() * 0.5);
      const rent = inputs.rent * (0.85 + Math.random() * 0.3);
      const totalRev = rev + inputs.wdfRevenue * (0.7 + Math.random() * 0.6);
      const totalOpEx = rent + inputs.utilities * (0.8 + Math.random() * 0.4) + inputs.labor * (0.9 + Math.random() * 0.2);
      const tpd = totalRev / (inputs.machines * 365);
      const rentRatio = rent / totalRev;

      const rentScore = Math.max(0, 100 - rentRatio * 1000);
      const tpdScore = Math.min(100, tpd * 20);
      const expenseScore = Math.max(0, 100 - (totalOpEx / totalRev) * 200);
      const score = Math.round((rentScore * 0.35) + (tpdScore * 0.40) + (expenseScore * 0.25));

      scores.push(score);
    }

    scores.sort((a, b) => a - b);
    const p10 = scores[Math.floor(simulations * 0.10)];
    const p50 = scores[Math.floor(simulations * 0.50)];
    const p90 = scores[Math.floor(simulations * 0.90)];

    setResults({ p10, p50, p90, avg: Math.round(scores.reduce((a, b) => a + b) / simulations) });
  };
