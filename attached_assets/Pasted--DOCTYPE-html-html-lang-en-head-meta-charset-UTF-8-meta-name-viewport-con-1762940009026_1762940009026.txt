    <script>
        let roiChart;
        function calculateROI() {
            const price = parseFloat(document.getElementById('purchasePrice').value);
            const rev = parseFloat(document.getElementById('monthlyRevenue').value) * 12;
            const exp = parseFloat(document.getElementById('monthlyExpenses').value) * 12;
            const cashFlow = rev - exp;
            const coc = (cashFlow / price) * 100;
            const breakEven = (price / cashFlow) * 12;
            const fiveYear = ((cashFlow * 5 - price) / price) * 100;

            document.getElementById('annualCashFlow').textContent = `$${cashFlow.toFixed(0)}`;
            document.getElementById('cashOnCash').textContent = `${coc.toFixed(1)}%`;
            document.getElementById('breakEven').textContent = `${breakEven.toFixed(0)}`;
            document.getElementById('fiveYearROI').textContent = `${Math.max(0, fiveYear).toFixed(1)}%`;

            const ctx = document.getElementById('roiChart').getContext('2d');
            if (roiChart) roiChart.destroy();
            roiChart = new Chart(ctx, {
                type: 'bar',
                data: { labels: ['Year 1', 'Year 3', 'Year 5'], datasets: [{ label: 'Cumulative ROI', data: [coc, coc*3, fiveYear], backgroundColor: '#007bff' }] },
                options: { responsive: true, scales: { y: { beginAtZero: true } } }
            });
        }
        document.querySelectorAll('input').forEach(el => el.addEventListener('input', calculateROI));
        window.onload = calculateROI;
    </script>
