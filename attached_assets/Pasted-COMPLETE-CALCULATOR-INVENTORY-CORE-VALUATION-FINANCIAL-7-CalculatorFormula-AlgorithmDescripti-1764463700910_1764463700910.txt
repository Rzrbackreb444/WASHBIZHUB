'use client';

import { useState } from 'react';
import { motion, AnimatePresence } from 'framer-motion';
import Confetti from 'react-confetti';
import { jsPDF } from 'jspdf';

interface Question {
  id: number;
  question: string;
  type: 'multiple' | 'open';
  options?: string[];
  correct: number | string;
  explanation: string;
}

interface ChapterQuizProps {
  chapter: number;
  questions: Question[];
  chapterTitle: string;
}

export default function BibleQuiz({ chapter, questions, chapterTitle }: ChapterQuizProps) {
  const [current, setCurrent] = useState(0);
  const [answers, setAnswers] = useState<(number | string)[]>([]);
  const [feedbacks, setFeedbacks] = useState<string[]>([]);
  const [showResult, setShowResult] = useState(false);
  const [score, setScore] = useState(0);
  const [showConfetti, setShowConfetti] = useState(false);

  const submitAnswer = async (answer: number | string) => {
    const q = questions[current];

    // AI Grading for open-ended (use Claude API)
    let fb = '';
    if (q.type === 'open') {
      const res = await fetch('/api/grade', {
        method: 'POST',
        body: JSON.stringify({ question: q.question, userAnswer: answer, correct: q.correct }),
      });
      const { feedback, score } = await res.json();
      fb = feedback;
      if (score >= 80) setShowConfetti(true);
    } else {
      fb = answer === q.correct ? 'Damn right!' : q.explanation;
    }

    setAnswers([...answers, answer]);
    setFeedbacks([...feedbacks, fb]);

    if (current < questions.length - 1) {
      setCurrent(current + 1);
    } else {
      const correctCount = answers.filter((a, i) => a === questions[i].correct).length;
      setScore((correctCount / questions.length) * 100);
      setShowResult(true);
      if (score >= 80) setShowConfetti(true);
    }
  };

  const generateCertificate = () => {
