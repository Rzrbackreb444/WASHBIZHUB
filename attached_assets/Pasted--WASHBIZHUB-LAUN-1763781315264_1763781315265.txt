===========================
MODULE 1 — CLEANBI v2.0
===========================
# Inputs:
# - Demographics (Renters%, income, density, growth)
# - Foot traffic & transit
# - TPD, EBITDA, Rev/SqFt, DSCR, SDE
# - Lease terms (rent%, term, TI, HOA)
# - Competition (washers, condition, distance, pricing)
# - Equipment (age, IoT faults, efficiency, replacement value)
# - Utilities (cost/turn, trend)
# - Digital readiness (web presence, reviews)
# - Risk (crime, flood, zoning)
# Output: CLEANBI_Score (0–100), Grade (A–F), Confidence Level

Weights:
MS=0.20; FS=0.20; LS=0.12; CS=0.12; ES=0.11; US=0.07; RS=0.08; SS=0.05; RA=-0.01

# Sample Subscore calculations:
RentersDensityScore = clamp((RentersPerSqMile / 1500) * 100, 0, 100)
MedianIncomeFitScore = clamp((1 - abs(MedianIncome-50000)/50000)*100,0,100)
FootTrafficScore = clamp((MonthlyVisits/1000)*10,0,100)
EquipmentScore = 0.5*(10-AvgMachineAge)/10*100 + 0.3*(1-MaintCostPct/0.2)100 + 0.2(PaymentSmart?100:60)
CompetitionScore = 100 - (CompetitionImpactDensity/BenchmarkCapacity)*100

CLEANBI = sum(weight * Subscore)

===========================
MODULE 5 — VALUATION ENGINE
===========================
# Core Financial Calculations
EBITDA = GrossRevenue - COGS - OpEx + Addbacks
SDE = EBITDA + OwnerCompensation
NOI = EBITDA - CAPEX

# Method 1: EBITDA Multiple
EBITDA_Multiple = if(CLEANBI>80,4.5, if(CLEANBI>70,3.8, if(CLEANBI>55,3.2,2.5)))
Value_EBITDA = EBITDA * EBITDA_Multiple

# Method 2: SDE Multiple
SDE_Multiple = if(CLEANBI>=80,3.0, if(CLEANBI>=60,2.5,2.0))
Value_SDE = SDE * SDE_Multiple

# Method 3: Asset Valuation
Value_Assets = EquipmentFMV + LeaseholdValue + Inventory - DebtOutstanding

# Method 4: Income Approach
CAP_RATE = 0.15  # default
Value_Income = NOI / CAP_RATE

# Final Weighted Valuation
Valuation = 0.35*Value_EBITDA + 0.25*Value_SDE + 0.20*Value_Income + 0.20*Value_Assets
