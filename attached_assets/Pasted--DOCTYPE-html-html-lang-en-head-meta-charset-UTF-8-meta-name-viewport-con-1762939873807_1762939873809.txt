        function matchFunding() {
            const amount = parseInt(document.getElementById('fundingAmount').value) || 0;
            const stage = document.getElementById('businessStage').value;
            const credit = document.getElementById('creditScore').value;
            const type = document.getElementById('fundingType').value;
            const location = document.getElementById('location').value.toUpperCase();

            let matches = fundingOptions.filter(option => {
                const amtMatch = amount >= (option.amount.includes('Up') ? 5000 : parseInt(option.amount.replace(/[^0-9]/g, '')));
                const stageMatch = option.stage.includes(stage);
                const creditMatch = option.credit.includes(credit) || option.credit === 'All' || option.credit === 'N/A';
                const typeMatch = !type || option.type === type || (type === 'loan' && option.type === 'loan');
                const stateMatch = !location || option.state === 'All' || option.state.includes(location);
                return amtMatch && stageMatch && creditMatch && typeMatch && stateMatch;
            });

            if (type === 'loan') matches = matches.filter(m => m.type === 'loan' || m.name.includes('SBA'));
            if (type === 'grant') matches = matches.filter(m => m.type === 'grant');

            matches.sort((a, b) => (a.type === type ? -1 : 1));
            matches = matches.slice(0, 6);

            const list = document.getElementById('matchList');
            if (matches.length === 0) {
                list.innerHTML = '<p style="text-align: center; font-style: italic;">No exact matches—broaden your criteria for more options!</p>';
            } else {
                list.innerHTML = matches.map(match => `
                    <div class="match-card">
                        <h4>${match.name}</h4>
                        <p>${match.desc}</p>
                        <p><strong>Amount:</strong> ${match.amount} | <strong>Eligibility:</strong> ${match.eligibility}</p>
                        <a href="${match.link}" target="_blank" rel="noopener">Apply Now</a>
                    </div>
                `).join('');
            }

            document.getElementById('results').style.display = 'block';
            window.scrollTo({ top: document.getElementById('results').offsetTop - 100, behavior: 'smooth' });
        }

        function applyNow() {
            const amount = document.getElementById('fundingAmount').value;
            const stage = document.getElementById('businessStage').options[document.getElementById('businessStage').selectedIndex].text;
            const matches = document.querySelectorAll('.match-card');
            if (matches.length > 0) {
                const topLink = matches[0].querySelector('a').href;
                alert(`Redirecting to top match for $${amount} ${stage.toLowerCase()}...\n\nPre-filled details ready.`);
                window.open(topLink, '_blank');
            } else {
                alert('Run a match first!');
            }
        }

        // Auto-match on change
        document.querySelectorAll('input, select').forEach(el => {
            el.addEventListener('change', matchFunding);
        });

        // Initial load with default
        window.onload = () => matchFunding();
