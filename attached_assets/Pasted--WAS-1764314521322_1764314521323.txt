'use client';
import { Canvas } from '@react-three/fiber';
import { OrbitControls, Grid, PerspectiveCamera } from '@react-three/drei';
import { useState } from 'react';
import SpeedQueen60 from './models/speedqueen-60.glb';
import DexterT900 from './models/dexter-t900.glb';
// ... 40+ more models

export default function DesignStudio() {
  const [machines, setMachines] = useState([]);
  const [revenue, setRevenue] = useState(0);
  const [cleanbi, setCleanbi] = useState(0);

  const addMachine = (model) => {
    setMachines([...machines, model]);
    setRevenue(prev => prev + model.monthlyRevenue);
    setCleanbi(prev => Math.min(100, prev + model.cleanbiBoost));
  };

  return (
    <div className="min-h-screen bg-navy-900 text-white">
      {/* Header */}
      <header className="fixed top-0 z-50 w-full bg-navy-900/95 backdrop-blur border-b border-teal-400/30">
        <div className="max-w-7xl mx-auto px-6 py-4 flex justify-between items-center">
          <h1 className="text-4xl font-bebas text-teal-400">WashBizHub 3D Design Studio</h1>
          <div className="text-2xl">
            Revenue: <span className="text-teal-400 font-bold">${revenue.toLocaleString()}/mo</span> · 
            CLEANBI: <span className="text-teal-400 font-bold">{cleanbi}/100</span>
          </div>
        </div>
      </header>

      <div className="flex h-screen pt-20">
        {/* 3D Canvas */}
        <div className="flex-1">
          <Canvas>
            <PerspectiveCamera makeDefault position={[20, 20, 20]} />
            <OrbitControls />
            <ambientLight intensity={0.8} />
            <directionalLight position={[10, 10, 5]} />
            <Grid infiniteGrid fadeDistance={50} />
            {/* Dropped machines render here */}
            {machines.map((m, i) => <primitive key={i} object={m.model} position={m.pos} />)}
          </Canvas>
        </div>

        {/* Sidebar */}
        <div className="w-96 bg-navy-800 p-8 overflow-y-auto">
          <h2 className="text-3xl font-bebas text-teal-400 mb-8">Drag Machines In</h2>
          <div className="grid grid-cols-2 gap-6">
            <button onClick={() => addMachine({ model: SpeedQueen60, monthlyRevenue: 18400, cleanbiBoost: 12 })} className="bg-teal-400/20 p-6 rounded-xl hover:bg-teal-400/40 transition">
              Speed Queen 60 lb · +$18,400/mo
            </button>
            {/* 40+ more buttons */}
          </div>

          <div className="mt-12 bg-teal-400/10 rounded-2xl p-8 text-center">
            <h3 className="text-4xl font-bold">Dynamic Pricing = +$31,200/year</h3>

              Launch This Store in WashBizPOS (Free Trial)
            </a>
          </div>

}
