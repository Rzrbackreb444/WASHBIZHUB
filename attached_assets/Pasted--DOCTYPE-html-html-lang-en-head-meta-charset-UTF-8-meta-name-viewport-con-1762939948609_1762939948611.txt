    <script>
        // Tab System
        function showTab(tab) {
            document.querySelectorAll('.calc-section').forEach(s => s.classList.remove('active'));
            document.getElementById(tab).classList.add('active');
            document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
            event.target.classList.add('active');
        }

        // ROI
        function calculateROI() {
            const price = parseFloat(document.getElementById('roi-purchase').value);
            const rev = parseFloat(document.getElementById('roi-revenue').value) * 12;
            const exp = parseFloat(document.getElementById('roi-expenses').value) * 12;
            const cash = rev - exp;
            const coc = (cash / price) * 100;
            const be = (price / cash) * 12;
            const five = ((cash * 5 - price) / price) * 100;

            document.getElementById('roi-cash').textContent = `$${cash.toFixed(0)}`;
            document.getElementById('roi-coc').textContent = `${coc.toFixed(1)}%`;
            document.getElementById('roi-break').textContent = `${be.toFixed(0)}`;
            document.getElementById('roi-5yr').textContent = `${Math.max(0, five).toFixed(1)}%`;
            updateROIChart(coc, five);
        }

        // Financing
        function calculateFinancing() {
            const cost = parseFloat(document.getElementById('fin-cost').value);
            const down = parseFloat(document.getElementById('fin-down').value);
            const months = parseInt(document.getElementById('fin-term').value);
            const rate = parseFloat(document.getElementById('fin-rate').value) / 100 / 12;
            const financed = cost - down;
            const monthly = financed * (rate * Math.pow(1 + rate, months)) / (Math.pow(1 + rate, months) - 1);
            const total = down + (monthly * months);
            const tco = total + 2500 + 4000; // maint + energy

            document.getElementById('fin-monthly').textContent = `$${monthly.toFixed(2)}`;
            document.getElementById('fin-total').textContent = `$${total.toFixed(2)}`;
            document.getElementById('fin-tco').textContent = `$${tco.toFixed(2)}`;
            updateFinChart(monthly, total - financed, 2500, 4000);
        }

        // Revenue
        function calculateRevenue() {
            const price = parseFloat(document.getElementById('rev-wash-price').value);
            const occ = parseFloat(document.getElementById('rev-wash-occ').value) / 100;
            const monthly = price * 10 * 30 * occ * 10; // 10 washers
            const annual = monthly * 12;

            document.getElementById('rev-monthly').textContent = `$${monthly.toFixed(0)}`;
            document.getElementById('rev-annual').textContent = `$${annual.toFixed(0)}`;
            updateRevChart(monthly);
        }

        // Charts
        let roiChart, finChart, revChart;
        function updateROIChart(coc, five) {
            const ctx = document.getElementById('roiChart').getContext('2d');
            if (roiChart) roiChart.destroy();
            roiChart = new Chart(ctx, { type: 'bar', data: { labels: ['Cash-on-Cash', '5-Year'], datasets: [{ data: [coc, five], backgroundColor: '#007bff' }] }, options: { responsive: true } });
        }
        function updateFinChart(monthly, interest, maint, energy) {
            const ctx = document.getElementById('finChart').getContext('2d');
            if (finChart) finChart.destroy();
            finChart = new Chart(ctx, { type: 'doughnut', data: { labels: ['P&I', 'Interest', 'Maint', 'Energy'], datasets: [{ data: [monthly*60, interest, maint, energy], backgroundColor: ['#007bff', '#dc3545', '#ffc107', '#28a745'] }] }, options: { responsive: true } });
        }
        function updateRevChart(monthly) {
            const ctx = document.getElementById('revChart').getContext('2d');
            if (revChart) revChart.destroy();
            revChart = new Chart(ctx, { type: 'line', data: { labels: ['Jan', 'Apr', 'Jul', 'Oct'], datasets: [{ label: 'Revenue', data: [monthly, monthly*1.05, monthly*1.1, monthly*1.15], borderColor: '#28a745' }] }, options: { responsive: true } });
        }

        // Universal PDF Export
        async function exportPDF(type) {
            const { jsPDF } = window.jspdf;
            const doc = new jsPDF();
            let y = 20;

            doc.setFontSize(20);
            doc.text(`WashBizHub ${type.toUpperCase()} Report`, 105, y, { align: "center" }); y += 15;
            doc.setFontSize(10);
            doc.text(`Generated: ${new Date().toLocaleString()}`, 105, y, { align: "center" }); y += 20;

            if (type === 'roi') {
                doc.setFontSize(14); doc.text("ROI Summary", 20, y); y += 10;
                doc.setFontSize(11);
                ['roi-cash', 'roi-coc', 'roi-break', 'roi-5yr'].forEach(id => {
                    doc.text(`${document.querySelector(`#${id}`).previousElementSibling.textContent} ${document.getElementById(id).textContent}`, 20, y); y += 7;
                });
            } else if (type === 'financing') {
                doc.setFontSize(14); doc.text("Financing + TCO", 20, y); y += 10;
                doc.setFontSize(11);
                ['fin-monthly', 'fin-total', 'fin-tco'].forEach(id => {
                    doc.text(`${document.querySelector(`#${id}`).previousElementSibling.textContent} ${document.getElementById(id).textContent}`, 20, y); y += 7;
                });
            } else if (type === 'revenue') {
                doc.setFontSize(14); doc.text("Revenue Forecast", 20, y); y += 10;
                doc.setFontSize(11);
                ['rev-monthly', 'rev-annual'].forEach(id => {
                    doc.text(`${document.querySelector(`#${id}`).previousElementSibling.textContent} ${document.getElementById(id).textContent}`, 20, y); y += 7;
                });
            }

            doc.setFontSize(9);
            doc.text("WashBizHub.com | Professional Laundromat Tools", 105, 280, { align: "center" });
            doc.save(`WashBizHub_${type}_Report_${new Date().toISOString().slice(0,10)}.pdf`);
        }

        // Auto-calculate
        document.querySelectorAll('input, select').forEach(el => el.addEventListener('input', () => {
            const active = document.querySelector('.calc-section.active').id;
            if (active === 'roi') calculateROI();
            else if (active === 'financing') calculateFinancing();
            else if (active === 'revenue') calculateRevenue();
        }));

        // Init
        window.onload = () => { calculateROI(); calculateFinancing(); calculateRevenue(); };
    </script>
