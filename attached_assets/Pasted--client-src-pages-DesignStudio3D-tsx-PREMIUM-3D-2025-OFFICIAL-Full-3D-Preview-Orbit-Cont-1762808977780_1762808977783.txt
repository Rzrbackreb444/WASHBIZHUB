// 3D Item Component
function Item3D({ item, isSelected, onSelect }) {
  const mesh = useRef();
  const [hovered, setHovered] = useState(false);

  return (
    <group position={[item.x * SCALE, item.height * SCALE / 2, item.z * SCALE]} onClick={() => onSelect(item.id)}>
      <Box
        ref={mesh}
        args={[item.width * SCALE, item.height * SCALE, item.depth * SCALE]}
        onPointerOver={() => setHovered(true)}
        onPointerOut={() => setHovered(false)}
      >
        <meshStandardMaterial color={hovered ? '#fbbf24' : item.color} />
      </Box>
      <Text
        position={[0, item.height * SCALE + 0.5, 0]}
        fontSize={0.8}
        color="#fff"
        anchorX="center"
        anchorY="middle"
      >
        {item.name}
      </Text>
      {isSelected && (
        <mesh position={[0, item.height * SCALE / 2, 0]}>
          <boxGeometry args={[item.width * SCALE + 0.2, item.height * SCALE + 0.2, item.depth * SCALE + 0.2]} />
          <meshBasicMaterial color="#fbbf24" wireframe />
        </mesh>
      )}
    </group>
  );
}

// Floating AI Tip
function AITip() {
  const mesh = useRef();
  useFrame((state) => {
    mesh.current.position.y = Math.sin(state.clock.elapsedTime) * 0.2 + 5;
  });

  return (
    <group ref={mesh} position={[0, 5, 0]}>
      <Box args={[8, 2, 0.1]}>
        <meshStandardMaterial color="#1a1a1a" />
      </Box>
      <Text position={[0, 0, 0.06]} fontSize={0.6} color="#fbbf24" anchorX="center">
        AI Tip: Move vending to corner = +$3K/mo
      </Text>
    </group>
  );
}
