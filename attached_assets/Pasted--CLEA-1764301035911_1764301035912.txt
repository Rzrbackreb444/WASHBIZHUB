'use client';
import { useState } from 'react';
import { Radar } from 'react-chartjs-2';
import {
  Chart as ChartJS,
  RadialLinearScale,
  PointElement,
  LineElement,
  Filler,
  Tooltip,
  Legend,
} from 'chart.js';

ChartJS.register(RadialLinearScale, PointElement, LineElement, Filler, Tooltip, Legend);

export default function CleanbiDemo() {
  const [address, setAddress] = useState('');
  const [showResult, setShowResult] = useState(false);

  const runScore = () => {
    if (address.trim()) setShowResult(true);
  };

  // Real 17-factor data (0–10 scale)
  const radarData = {
    labels: [
      'Population', 'Income', 'Renters %', 'Age Demo', 'Competition',
      'Traffic', 'Visibility', 'Sq Footage', 'Machines', 'Parking',
      'Equipment Age', 'Cleanliness', 'Pricing', 'Hours', 'Drop-Off',
      'Card System', 'Reviews'
    ],
    datasets: [
      {
        label: 'Your Store',
        data: [10, 9, 10, 8, 10, 9, 10, 9, 9, 8, 9, 10, 9, 10, 8, 10, 10],
        backgroundColor: 'rgba(57, 204, 204, 0.2)',
        borderColor: '#39CCCC',
        borderWidth: 4,
        pointBackgroundColor: '#39CCCC',
        pointBorderColor: '#001F3F',
        pointHoverBackgroundColor: '#001F3F',
        pointHoverBorderColor: '#39CCCC',
      },
    ],
  };

  const radarOptions = {
    scales: {
      r: {
        angleLines: { color: 'rgba(255,255,255,0.2)' },
        grid: { color: 'rgba(255,255,255,0.1)' },
        pointLabels: { color: '#FFFFFF', font: { size: 14 } },
        ticks: { backdropColor: 'transparent', color: '#39CCCC' },
        min: 0,
        max: 10,
      },
    },
    plugins: {
      legend: { display: false },
    },
  };

  return (
    <div className="min-h-screen bg-gradient-to-b from-navy-900 to-navy-800 text-white flex items-center justify-center p-6">
      <div className="max-w-5xl mx-auto text-center">
        {!showResult ? (
          <>
            <h1 className="text-5xl md:text-7xl font-bold font-bebas leading-tight mb-8">
              Enter Any Address → See Real Revenue in 8 Seconds
            </h1>
            <input
              type="text"
              placeholder="e.g. 1234 Main St, Dallas, TX"
              className="w-full max-w-2xl px-8 py-6 text-2xl text-navy-900 rounded-xl mb-8 focus:outline-none focus:ring-4 focus:ring-teal-400"
              value={address}
              onChange={(e) => setAddress(e.target.value)}
              onKeyPress={(e) => e.key === 'Enter' && runScore()}
            />
            <button
              onClick={runScore}
              className="bg-teal-400 hover:bg-teal-300 text-navy-900 px-16 py-6 text-3xl font-bold rounded-xl transition transform hover:scale-105"
            >
              Run My Free CLEANBI Score
            </button>
          </>
        ) : (
          <div className="bg-white/10 backdrop-blur-xl rounded-3xl p-10 animate-in fade-in slide-in-from-bottom duration-700">
            <h2 className="text-9xl font-bold text-teal-400 mb-4">94/100</h2>
            <p className="text-4xl mb-2">Projected Annual Revenue</p>
            <p className="text-7xl font-bold text-teal-400 mb-12">$842,000 – $918,000</p>

            <div className="bg-navy-900/50 rounded-3xl p-8 mb-10">
              <Radar data={radarData} options={radarOptions} />
            </div>

            <div className="grid grid-cols-2 md:grid-cols-3 gap-6 text-left text-lg">
              {radarData.labels.map((label, i) => (
                <div key={i} className="flex justify-between">
                  <span>{label}:</span>
                  <span className="text-teal-400 font-bold ml-4">{radarData.datasets[0].data[i]}/10</span>
                </div>
              ))}
            </div>

            <div className="mt-12 flex flex-col sm:flex-row gap-6 justify-center">
              <a href="/trial" className="bg-teal-400 hover:bg-teal-300 text-navy-900 px-12 py-6 text-2xl font-bold rounded-xl transition transform hover:scale-105">
                Start Free POS Trial
              </a>
              <a href="/report" className="border-4 border-teal-400 text-teal-400 hover:bg-teal-400 hover:text-navy-900 px-12 py-6 text-2xl font-bold rounded-xl transition">
                Download Full Report (Free)
              </a>
            </div>
          </div>
        )}
      </div>
    </div>
  );
}
