// Assume a rate limiter is configured for 500 requests per day (Enterprise limit)
const ENTERPRISE_LIMIT = 500;

async function checkRateLimit(userId: string): Promise<boolean> {
  // Use Redis or similar to track requests for this userId
  const usage = await redis.incrementUsage(userId, 'daily');

C. API Route Handler (/api/broker/cleanbi)This is the main Next.js API route that ties the logic together.TypeScript// pages/api/broker/cleanbi.ts (or app/api/broker/cleanbi/route.ts)

import type { NextApiRequest, NextApiResponse } from 'next';
// Import helper functions defined above
import { verifyBrokerAccess } from 'lib/auth';
import { checkRateLimit } from 'lib/rateLimiter';
import { getCleanbiData } from 'lib/cleanbi'; // Core proprietary logic

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  const apiKey = req.headers['x-api-key'] as string;
  const { address } = req.query;

  // 1. Authorization Check (401/403)
  const user = await verifyBrokerAccess(apiKey);
  if (!user) {
    return res.status(401).json({ message: 'Unauthorized: Invalid or insufficient API Key.' });
  }

  // 2. Rate Limit Check (429)
  const isRateLimited = await checkRateLimit(user.userId);
  if (isRateLimited) {
    return res.status(429).json({ message: 'Too many requests. Limit exceeded.' });
  }

  // 3. Process Request (200)
  try {
    const cleanbiReport = await getCleanbiData(address as string);
    // Return the simplified, structured data brokers need
    return res.status(200).json({ 
      cleanbiScore: cleanbiReport.score, 
      grade: cleanbiReport.grade, 
      demographics: cleanbiReport.key_factors 
    });
  } catch (error) {
    console.error(error);
    return res.status(500).json({ message: 'Internal Server Error during data generation.' });
  }
}
