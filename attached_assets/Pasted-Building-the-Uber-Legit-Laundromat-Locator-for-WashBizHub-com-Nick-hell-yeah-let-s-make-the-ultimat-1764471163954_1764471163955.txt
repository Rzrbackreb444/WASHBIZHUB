function App() {
  const [search, setSearch] = useState('');
  const [filter, setFilter] = useState('all'); // all, 24hr, coin, card, for_sale
  const [map, setMap] = useState(null);
  const [selected, setSelected] = useState(null);

  const filtered = laundromats.filter(l => 
    l.name.toLowerCase().includes(search.toLowerCase()) && 
    (filter === 'all' || l[filter])
  );

  const LocationMarker = () => {
    useMapEvents({
      click(e) {
        // Add new laundromat pin
        const newLaund = { id: Date.now(), lat: e.latlng.lat, lng: e.latlng.lng, name: 'New Location', cleanbi: 75, revenue_est: 500000 };
        setSelected(newLaund);
      },
    });
    return null;
  };

  const handlePOSUpsell = (laund) => {
    // Integrate your POS trial
    window.location.href = `/trial?laund=${laund.id}`;
  };

  return (
    <div className="min-h-screen bg-gray-100">
      {/* Header */}
      <header className="bg-white shadow-lg p-4">
        <div className="max-w-6xl mx-auto flex justify-between items-center">
          <h1 className="text-3xl font-bold text-blue-600">WashBizHub Laundromat Locator</h1>
          <div className="flex gap-4">
            <input
              type="text"
              placeholder="Search by name or city..."
              value={search}
              onChange={(e) => setSearch(e.target.value)}
              className="px-4 py-2 border rounded-lg"
            />
            <select value={filter} onChange={(e) => setFilter(e.target.value)} className="px-4 py-2 border rounded-lg">
              <option value="all">All</option>
              <option value="24hr">24-Hour</option>
              <option value="coin">Coin-Op</option>
              <option value="card">Card Pay</option>
              <option value="for_sale">For Sale</option>
            </select>
          </div>
        </div>
      </header>

      {/* Map */}
      <div className="h-96">
        <MapContainer center={[39.8283, -98.5795]} zoom={4} style={{ height: '100%', width: '100%' }} whenCreated={setMap}>
          <TileLayer url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png" attribution='&copy; OpenStreetMap' />
          <LocationMarker />
          {filtered.map(laund => (
            <Marker key={laund.id} position={[laund.lat, laund.lng]}>
              <Popup>
                <h3>{laund.name}</h3>
                <p>{laund.address}, {laund.city}, {laund.state} {laund.zip}</p>
                <p>Rating: {laund.rating} ({laund.reviews} reviews)</p>
                <p>CLEANBI Score: {laund.cleanbi}/100</p>
                <p>Est. Annual Revenue: ${laund.revenue_est.toLocaleString()}</p>
                {laund.email && <p>Email: {laund.email}</p>}
                {laund.phone && <p>Phone: {laund.phone}</p>}
                {laund.website && <a href={laund.website} target="_blank">Website</a>}
                {laund.for_sale && <button onClick={() => handlePOSUpsell(laund)}>For Sale — Get Offer</button>}
                <button onClick={() => handlePOSUpsell(laund)}>Upgrade with WashBizPOS</button>
              </Popup>
            </Marker>
          ))}
        </MapContainer>
      </div>

      {/* List View */}
      <div className="max-w-6xl mx-auto p-4">
        <h2 className="text-2xl font-bold mb-4">Results ({filtered.length})</h2>
        {filtered.map(laund => (
          <div key={laund.id} className="bg-white p-4 mb-4 rounded-lg shadow">
            <h3 className="text-xl font-bold">{laund.name}</h3>
            <p>{laund.address}, {laund.city}, {laund.state} {laund.zip}</p>
            <p>Rating: {laund.rating} ({laund.reviews} reviews)</p>
            <p>CLEANBI: {laund.cleanbi}/100 | Est. Revenue: ${laund.revenue_est.toLocaleString()}/yr</p>
            <button onClick={() => handlePOSUpsell(laund)}>Upgrade with POS</button>
          </div>
        ))}
      </div>
    </div>
  );
}
